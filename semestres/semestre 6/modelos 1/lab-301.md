import itertools


def fit(X,y):

  def predict(X, t):
    return ((X[:,0] >= t[0]) | (X[:,1] <= t[1])).astype(int)
	
  possible_values = np.linspace(0.0,1.0,num=11)
  best_pair = None
  best_result = 0
  for pair in itertools.product(possible_values, repeat=2):
    y_cap = predict(X, pair)
    result = np.mean(y_cap==y)
    if best_result < result:
      best_pair, best_result = pair, result
      
  return np.array(best_pair)
