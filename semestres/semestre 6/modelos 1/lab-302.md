```
def make_timeseries_dataset(signal, n_timesteps_lookback):
    import pandas as pd
    r = signal
    for i in range(n_timesteps_lookback):
      other = signal.shift(periods=i+1)
      r = r.join(other, lsuffix='', rsuffix='-'+str(i+1) )
    
    other = signal.shift(periods=-1)
    r = r.join(other, lsuffix='', rsuffix='+1')

    return r.dropna()
```

```
def apply_linear_regression_model(td, w):
    r = w[0] + td['signal'] * w[1] + sum(td[f'signal-{i+1}'] * w[i+2] for i in range(len(w)-2))
    return r.values
```

```
def measure_trend_accuracy(td, preds):
    import numpy as np
    r = np.mean(((td['signal+1'] > td['signal']) & (preds > td['signal'])) | ((td['signal+1'] < td['signal']) & (preds < td['signal'])))
    return r
```
